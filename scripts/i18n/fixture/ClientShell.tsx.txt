"use client";
import createCache from "@emotion/cache";
import { CacheProvider } from "@emotion/react";
import { setupI18n } from "@lingui/core";
import { I18nProvider, useLingui } from "@lingui/react";
import { Trans } from "@lingui/react/macro";
import { Box, Button, CssBaseline, TextField } from "@mui/material";
import { createTheme, ThemeProvider } from "@mui/material/styles";
import { AppRouterCacheProvider } from "@mui/material-nextjs/v16-appRouter";
import rtlPlugin from "@mui/stylis-plugin-rtl";
import { prefixer } from "stylis";
import { useLayoutEffect, useState } from "react";
import { catalogs } from "./messages";

const options = (locale) => ({
  key: locale === "he" ? "proof-rtl" : "proof-ltr",
  enableCssLayer: true,
  stylisPlugins: locale === "he" ? [prefixer, rtlPlugin] : [prefixer],
});
const themes = {
  en: createTheme({ direction: "ltr" }),
  he: createTheme({ direction: "rtl" }),
};

function DirectionCache({ value, children }) {
  const { i18n } = useLingui();
  const [caches] = useState(() => new Map([[i18n.locale, value]]));
  if (!caches.has(i18n.locale)) {
    // The first cache is streamed by MUI. Only a subsequent client switch
    // creates this cache; retain the CSS layer without remounting children.
    const cache = createCache(options(i18n.locale));
    const insert = cache.insert;
    cache.insert = (...args) => {
      if (!args[1].styles.startsWith("@layer"))
        args[1].styles = `@layer mui {${args[1].styles}}`;
      return insert(...args);
    };
    caches.set(i18n.locale, cache);
  }
  return (
    <CacheProvider value={caches.get(i18n.locale)}>{children}</CacheProvider>
  );
}
function Content({ editor, children }) {
  const { i18n } = useLingui();
  const [draft, setDraft] = useState("Unsaved synthetic draft");
  useLayoutEffect(() => {
    document.documentElement.lang = i18n.locale;
    document.documentElement.dir = i18n.locale === "he" ? "rtl" : "ltr";
  }, [i18n.locale]);
  return (
    <ThemeProvider theme={themes[i18n.locale]}>
      <CssBaseline />
      <Box id="direction-probe" sx={{ ml: "17px" }}>
        <Trans>Unsaved draft</Trans>
      </Box>
      {editor && (
        <>
          <TextField
            id="draft"
            multiline
            value={draft}
            onChange={(event) => setDraft(event.target.value)}
          />
          <Button
            id="switch-locale"
            onClick={() => i18n.activate(i18n.locale === "en" ? "he" : "en")}
          >
            Switch language
          </Button>
          <a id="view" href="/doc/public">
            View ritual
          </a>
        </>
      )}
      {children}
    </ThemeProvider>
  );
}
export default function ClientShell({ locale, editor = false, children }) {
  const [i18n] = useState(() => setupI18n({ locale, messages: catalogs }));
  return (
    <I18nProvider i18n={i18n}>
      <AppRouterCacheProvider
        options={options(locale)}
        CacheProvider={DirectionCache}
      >
        <Content editor={editor}>{children}</Content>
      </AppRouterCacheProvider>
    </I18nProvider>
  );
}
